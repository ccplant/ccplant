package services

import (
	"context"
	corev1 "k8s.io/api/core/v1"
	apierrors "k8s.io/apimachinery/pkg/api/errors"
	metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"
	"net/http"
	"net/http/httptest"
	"testing"
)

func TestInvestigationPurgeClaimingWindow(t *testing.T) {
	for _, protected := range []bool{false, true} {
		t.Run(map[bool]string{false: "claiming omitted", true: "claiming protected"}[protected], func(t *testing.T) {
			m := newWorkloadTestManager(t, false)
			ctx := context.Background()
			parent := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
				if protected {
					_, _ = w.Write([]byte(`{"allocated_runner_ids":["runner"]}`))
				} else {
					_, _ = w.Write([]byte(`{"allocated_runner_ids":[]}`))
				}
			}))
			defer parent.Close()
			m.ConfigureSessionRunnerPool(parent.URL, "manager", "test-token", "pool")
			labels := map[string]string{"app.kubernetes.io/name": "agentapi-session", "app.kubernetes.io/managed-by": "agentapi-proxy", "agentapi.proxy/session-id": "runner", "agentapi.proxy/session-pool": "pool", "agentapi.proxy/stock": "true"}
			_, err := m.client.CoreV1().Services("test-ns").Create(ctx, &corev1.Service{ObjectMeta: metav1.ObjectMeta{Name: "runner-svc", Labels: labels}}, metav1.CreateOptions{})
			if err != nil {
				t.Fatal(err)
			}
			if err := m.PurgeStockSessions(ctx); err != nil {
				t.Fatal(err)
			}
			_, err = m.client.CoreV1().Services("test-ns").Get(ctx, "runner-svc", metav1.GetOptions{})
			if protected && err != nil {
				t.Fatal(err)
			}
			if !protected && !apierrors.IsNotFound(err) {
				t.Fatalf("expected purge: %v", err)
			}
			t.Logf("parent protection=%v, service removed=%v", protected, apierrors.IsNotFound(err))
		})
	}
}
